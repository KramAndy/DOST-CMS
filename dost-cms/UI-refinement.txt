  function $(s) { return document.querySelector(s); }
  function el(html) { const t = document.createElement('template'); t.innerHTML = html.trim(); return t.content.firstChild; }

  function toast(msg, type = 'info') {
    const colors = { info: 'bg-[var(--ink)]', warn: 'bg-red-500', success: 'bg-[var(--accent-dark)]' };
    const icons = { info: 'fa-circle-info', warn: 'fa-triangle-exclamation', success: 'fa-circle-check' };
    const t = el(`<div class="toast ${colors[type]} text-white text-sm px-4 py-3 rounded-lg shadow-lg flex items-center gap-2 max-w-xs"><i class="fa-solid ${icons[type]}"></i>${msg}</div>`);
    $('#toast-container').appendChild(t);
    setTimeout(() => { t.style.opacity = '0'; t.style.transform = 'translateY(10px)'; t.style.transition = 'all 0.3s'; }, 2800);
    setTimeout(() => t.remove(), 3200);
  }

  function formatDate(dateStr) {
    const d = new Date(dateStr);
    return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
  }

  function getTimestamp() {
    const d = new Date();
    return d.toLocaleString('en-US', { month: 'short', day: 'numeric', year: 'numeric', hour: 'numeric', minute: '2-digit', hour12: true });
  }

  // === REFERENCE NUMBER GENERATION LOGIC ===
  function generateRefNumber(type) {
    const year = new Date().getFullYear();
    const sep = refFormat.separator;
    let prefix = refFormat.prefix;
    
    if (refFormat.includeYear) {
      prefix = `${prefix}${sep}${year}`;
    }
    
    const fullPrefix = `${prefix}${sep}${type}${sep}`;
    
    const typeDocs = documents.filter(d => d.ref.startsWith(fullPrefix));
    const maxSeq = typeDocs.reduce((max, d) => {
      const seq = parseInt(d.ref.split(sep).pop(), 10);
      return seq > max ? seq : max;
    }, 0);
    
    const newSeq = (maxSeq + 1).toString().padStart(refFormat.padding, '0');
    return `${fullPrefix}${newSeq}`;
  }

  function updateRefPreview() {
    const type = $('#doc-type-select').value;
    $('#ref-preview').textContent = generateRefNumber(type);
  }

  // === TABLE RENDERING (No full re-render on expand) ===
  function renderTable() {
    const tbody = $('#doc-tbody');
    const emptyState = $('#empty-state');
    tbody.innerHTML = '';
    
    let filtered = documents;
    if (currentFilter !== 'ALL') {
      filtered = filtered.filter(d => d.type === currentFilter);
    }
    if (currentYear !== 'ALL') {
      filtered = filtered.filter(d => new Date(d.date).getFullYear().toString() === currentYear);
    }
    if (currentMonth !== 'ALL') {
      filtered = filtered.filter(d => new Date(d.date).getMonth().toString() === currentMonth);
    }
    if (searchQuery) {
      const q = searchQuery.toLowerCase();
      filtered = filtered.filter(d => 
        d.ref.toLowerCase().includes(q) || 
        d.title.toLowerCase().includes(q) ||
        d.origin.toLowerCase().includes(q)
      );
    }

    if (filtered.length === 0) {
      emptyState.classList.remove('hidden');
    } else {
      emptyState.classList.add('hidden');
      
      filtered.slice().reverse().forEach(doc => {
        const statusClass = doc.status === 'Received' ? 'status-received' 
                          : doc.status === 'Processing' ? 'status-processing' 
                          : 'status-dispatched';
        
        const row = el(`
          <tr class="doc-row animate-slide-up" data-ref="${doc.ref}">
            <td class="px-6 py-4 align-top w-1/4">
              <div class="font-mono text-xs font-bold text-[var(--ink)]">${doc.ref}</div>
              <span class="text-[10px] font-bold px-2 py-0.5 rounded mt-1 inline-block border ${docTypes[doc.type].color}">${doc.type}</span>
            </td>
            <td class="px-6 py-4 align-top">
              <div class="font-semibold text-[13px] text-[var(--ink)] leading-tight">${doc.title}</div>
              <div class="text-[11px] text-[var(--muted)] mt-1"><i class="fa-solid fa-building text-[9px] mr-1"></i>${doc.origin}</div>
            </td>
            <td class="px-6 py-4 align-top hidden md:table-cell">
              <div class="text-xs text-[var(--muted)] font-medium whitespace-nowrap">${formatDate(doc.date)}</div>
            </td>
            <td class="px-6 py-4 align-top">
              <span class="status-badge ${statusClass}">${doc.status}</span>
            </td>
            <td class="px-6 py-4 align-top text-right">
              <button class="text-[var(--muted)] hover:text-[var(--accent)] transition-colors p-2 expand-icon">
                <i class="fa-solid fa-chevron-down"></i>
              </button>
            </td>
          </tr>
        `);
        tbody.appendChild(row);
      });
    }

    // Bind row clicks
    document.querySelectorAll('.doc-row').forEach(row => {
      row.addEventListener('click', (e) => {
        const ref = row.dataset.ref;
        toggleExpand(ref);
      });
    });

    // Re-open previously expanded row if it still exists
    if (expandedRef) {
      const row = document.querySelector(`.doc-row[data-ref="${expandedRef}"]`);
      if (row) openRow(row, expandedRef);
    }
  }

  // === SURGICAL EXPANSION LOGIC ===
  function toggleExpand(ref) {
    const row = document.querySelector(`.doc-row[data-ref="${ref}"]`);
    if (!row) return;
    
    const isExpanded = row.classList.contains('expanded');
    
    // Close any currently open rows
    document.querySelectorAll('.doc-row.expanded').forEach(r => {
      if (r !== row) closeRow(r);
    });

    if (isExpanded) {
      closeRow(row);
      expandedRef = null;
    } else {
      openRow(row, ref);
      expandedRef = ref;
    }
  }

  function closeRow(row) {
    row.classList.remove('expanded');
    const icon = row.querySelector('.expand-icon i');
    if (icon) icon.className = 'fa-solid fa-chevron-down';
    
    const expandRow = row.nextElementSibling;
    if (expandRow && expandRow.classList.contains('row-expand-wrapper')) {
      expandRow.remove();
    }
  }

  function openRow(row, ref) {
    row.classList.add('expanded');
    const icon = row.querySelector('.expand-icon i');
    if (icon) icon.className = 'fa-solid fa-chevron-up';
    
    const doc = documents.find(d => d.ref === ref);
    const safeRef = ref.replace(/[^a-zA-Z0-9]/g, '');
    
    const expandRow = el(`
      <tr class="row-expand-wrapper">
        <td colspan="5" class="p-0">
          <div class="row-expand-content">
            <div class="p-6 grid md:grid-cols-2 gap-8">
              <div>
                <h4 class="font-display font-bold text-sm mb-5 flex items-center gap-2 text-[var(--ink)]">
                  <i class="fa-solid fa-clock-rotate-left text-[var(--accent-dark)]"></i> Chronological Updates
                </h4>
                <div id="timeline-container-${safeRef}" class="px-2">
                  ${doc.history.map((h, idx) => `
                    <div class="timeline-item ${idx === 0 ? 'active' : ''} ${h.status === 'Dispatched' ? 'dark' : ''}">
                      <div class="text-[10px] text-[var(--muted)] font-bold uppercase">${h.ts}</div>
                      <div class="font-semibold text-sm mt-1 text-[var(--ink)]">${h.status}</div>
                      <p class="text-xs text-[var(--muted)] mt-1 leading-relaxed">${h.note}</p>
                    </div>
                  `).join('')}
                </div>
              </div>
              <div>
                <h4 class="font-display font-bold text-sm mb-5 flex items-center gap-2 text-[var(--ink)]">
                  <i class="fa-solid fa-pen-to-square text-[var(--accent-dark)]"></i> Provide Update
                </h4>
                <div class="mb-4">
                  <label class="text-[10px] uppercase tracking-wider text-[var(--muted)] font-bold">Quick Status Presets</label>
                  <div class="flex flex-wrap gap-2 mt-2">
                    <button class="btn-ghost text-xs font-semibold px-3 py-1.5 rounded-lg preset-btn" data-ref="${doc.ref}" data-status="Received">Received</button>
                    <button class="btn-ghost text-xs font-semibold px-3 py-1.5 rounded-lg preset-btn" data-ref="${doc.ref}" data-status="Processing">Processing</button>
                    <button class="btn-ghost text-xs font-semibold px-3 py-1.5 rounded-lg preset-btn" data-ref="${doc.ref}" data-status="Dispatched">Dispatched</button>
                  </div>
                </div>
                <div>
                  <label class="text-[10px] uppercase tracking-wider text-[var(--muted)] font-bold">Update Notes</label>
                  <textarea id="update-notes-${safeRef}" rows="3" placeholder="Enter details about this update..." class="input-flat mt-2 w-full px-4 py-2.5 rounded-lg text-sm resize-none"></textarea>
                </div>
                <button class="btn-primary w-full text-xs font-semibold px-4 py-2.5 rounded-lg flex items-center justify-center gap-2 mt-3 submit-update-btn" data-ref="${doc.ref}">
                  <i class="fa-solid fa-paper-plane text-[10px]"></i> Submit Update
                </button>
              </div>
            </div>
          </div>
        </td>
      </tr>
    `);
    
    row.after(expandRow);
    bindExpandEvents(expandRow, ref);
  }

  function bindExpandEvents(expandRow, ref) {
    const safeRef = ref.replace(/[^a-zA-Z0-9]/g, '');
    
    expandRow.querySelectorAll('.preset-btn').forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        const status = btn.dataset.status;
        const doc = documents.find(d => d.ref === ref);
        if (doc) {
          const newEntry = { ts: getTimestamp(), status: status, note: `Status automatically updated to ${status}.` };
          doc.status = status;
          doc.history.unshift(newEntry);
          
          // Surgically update UI
          updateDocStatusUI(ref, status);
          addTimelineItem(ref, newEntry);
          
          toast(`Status updated to ${status}`, 'success');
        }
      });
    });

    const submitBtn = expandRow.querySelector('.submit-update-btn');
    if (submitBtn) {
      submitBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        const doc = documents.find(d => d.ref === ref);
        const notesArea = expandRow.querySelector(`#update-notes-${safeRef}`);
        if (doc && notesArea && notesArea.value.trim()) {
          const newEntry = { ts: getTimestamp(), status: doc.status, note: notesArea.value.trim() };
          doc.history.unshift(newEntry);
          notesArea.value = '';
          
          addTimelineItem(ref, newEntry);
          toast('Update added to timeline', 'success');
        } else if (doc && notesArea && !notesArea.value.trim()) {
          toast('Please enter update notes', 'warn');
        }
      });
    }
  }

  function updateDocStatusUI(ref, status) {
    const row = document.querySelector(`.doc-row[data-ref="${ref}"]`);
    if (!row) return;
    const badge = row.querySelector('.status-badge');
    if (badge) {
      badge.className = `status-badge ${status === 'Received' ? 'status-received' : status === 'Processing' ? 'status-processing' : 'status-dispatched'}`;
      badge.textContent = status;
    }
  }

  function addTimelineItem(ref, item) {
    const safeRef = ref.replace(/[^a-zA-Z0-9]/g, '');
    const container = document.getElementById(`timeline-container-${safeRef}`);
    if (!container) return;
    
    // Remove 'active' from old first item
    const oldActive = container.querySelector('.timeline-item.active');
    if (oldActive) oldActive.classList.remove('active');
    
    const newItem = el(`
      <div class="timeline-item active ${item.status === 'Dispatched' ? 'dark' : ''}">
        <div class="text-[10px] text-[var(--muted)] font-bold uppercase">${item.ts}</div>
        <div class="font-semibold text-sm mt-1 text-[var(--ink)]">${item.status}</div>
        <p class="text-xs text-[var(--muted)] mt-1 leading-relaxed">${item.note}</p>
      </div>
    `);
    container.prepend(newItem);
  }
